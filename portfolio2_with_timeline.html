# CutIt — 2D Bin Packing · Karan N. Londe
import numpy as np
from dataclasses import dataclass
from typing import List, Tuple

@dataclass
class Rect:
    w: float; h: float
    x: float = 0.0; y: float = 0.0

class SheetOptimiser:
    def __init__(self, W: float, H: float):
        self.W, self.H = W, H
        self.spaces: List[Rect] = [Rect(W, H)]
        self.placed: List[Rect] = []

    def pack(self, pieces: List[Rect]) -> float:
          # Sort by area desc — best-fit heuristic
        pieces.sort(key=lambda p: p.w*p.h, reverse=True)
        for p in pieces:
            s = self._best_space(p)
            if s: self._place(p, s)
        used = sum(r.w*r.h for r in self.placed)
        return round(used / (self.W*self.H) * 100, 2)

    def _best_space(self, p):
        return min(
            (s for s in self.spaces
             if s.w >= p.w and s.h >= p.h),
            key=lambda s: s.w*s.h, default=None)
